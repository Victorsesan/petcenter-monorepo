## What Was Required

Task 3: let a visitor browse pets by location, reusing the existing Bottom Sheet + Filter Button + Search Bar pattern instead of adding a separate filter UI. It uses only free tools and was researched against how Petfinder does it. It covers a manual State and City filter, plus a way to see pets near the visitor or within a chosen radius of a place they type. Files: the seven category pages and `pets.html`.

Task 3b: extend the same filter to the confirmed target markets (USA, Canada, UK, Australia) with a country selector that defaults to the visitor's own country, one free geocoding service for all four countries, and the existing distance calculation reused. The CEO also told me to build the Recommended Enhancements (a "Use my location" button, a server-side distance calculation, and cached lookups), so they are included.

What was built: a Location group in the filter sheet (State chips, City chips, a Near Me box with country chips, a "Use my location" button and 25/60/100/250/500 mile chips), a "Nearest First" sort chip, a visible notice above the pet grid, and on `pets.html` a small state box whose value is carried into the category pages as `?state=`. Distance search runs on the server in a new endpoint, `geo-filter.php`, which geocodes pet locations through OpenStreetMap Nominatim (free, no key) into a shared cache file and sorts by distance. The visitor's own typed place is geocoded in the browser.

// TASK-3 START — Location filter helpers: geocoding + distance math.
// Free, no-API-key geocoding via OpenStreetMap Nominatim (usage policy caps
// requests at ~1/sec, so calls are chained through a single shared queue
// rather than fired in parallel). Results are cached in memory for this page
// load only — a persistent shared cache is a Task 3b enhancement, not needed
// here since a single visitor rarely repeats the same search.
var pcGeocodeCache = {};
var pcGeocodeQueue = Promise.resolve();
var pcLastGeocodeAt = 0; // TASK-3d — timestamp of the last Nominatim call, so a lone search isn't delayed a full second for nothing

function pcGeocodeLocation(query, countryCode) {
  var trimmed = (query || '').trim();
  if (!trimmed) return Promise.resolve(null);
  var key = trimmed.toLowerCase() + '|' + (countryCode || 'us'); // TASK-3b — cache key now includes country so the same text in two countries doesn't collide
  if (Object.prototype.hasOwnProperty.call(pcGeocodeCache, key)) {
    return Promise.resolve(pcGeocodeCache[key]);
  }
  pcGeocodeQueue = pcGeocodeQueue.then(function() {
    return new Promise(function(resolve) {
      setTimeout(function() {
        pcLastGeocodeAt = Date.now(); // TASK-3d
        // TASK-3b — countrycodes now driven by the selected country (us/ca/gb/au)
        // instead of being hardcoded to "us", covering all 4 confirmed target
        // markets through this same single Nominatim integration.
        fetch('https://nominatim.openstreetmap.org/search?format=json&limit=1&countrycodes=' + (countryCode || 'us') + '&q=' + encodeURIComponent(query))
          .then(function(r) { return r.ok ? r.json() : []; })
          .then(function(results) {
            var coords = (results && results[0])
              ? { lat: parseFloat(results[0].lat), lng: parseFloat(results[0].lon) }
              : null;
            pcGeocodeCache[key] = coords;
            resolve(coords);
          })
          .catch(function() {
            pcGeocodeCache[key] = null;
            resolve(null);
          });
      }, Math.max(0, 1100 - (Date.now() - pcLastGeocodeAt))); // TASK-3d — still never closer than ~1.1s apart (Nominatim's 1 req/sec policy), but no wait when the last call was long ago
    });
  });
  return pcGeocodeQueue;
}

// pet.json stores location as free text, e.g. "Los Angeles, CA"
function pcParseLocation(loc) {
  if (!loc) return { city: '', state: '' };
  var text = String(loc).trim();
  var parts = text.split(',');
  var city = (parts[0] || '').trim();
  var state = (parts[1] || '').trim();
  // TASK-3d START — pet.location is free text typed by an admin. Tolerate a missing comma
  // ("Los Angeles CA") by peeling a trailing 2-letter code off as the state, and upper-case
  // short codes so "ca" / "Ca" / "CA" never become three separate State chips.
  if (!state) {
    var m = text.match(/^(.*\S)\s+([A-Za-z]{2})$/);
    if (m) { city = m[1].trim(); state = m[2]; }
  }

// TASK-3c START — Calls geo-filter.php with the visitor's resolved
// coordinates + chosen radius; the server geocodes every pet in this
// category (using its own persisted cache) and returns just the IDs within
// range, nearest-first. Replaces Task 3b's pcGeocodeAllPetLocations(), which
// did this same work in the browser with a cache that reset every page load.
var pcRadiusMatchIds = null; // null = no server radius search active; array = active, filter to these IDs only
// TASK-3d START — second verification round hardening of the search helper:
//  - newest-search-wins token, so a slow earlier request can never overwrite a later one
//  - real failure handling: a missing/erroring endpoint no longer looks like "zero pets nearby"
//  - coordinates rounded to 2 decimals (~0.7 mi) before they go into the URL, because query
//    strings end up in server access logs and a visitor's exact position doesn't belong there;
//    the smallest radius offered is 25 mi, so nothing is lost
var pcRadiusSearchSeq = 0;
function pcRunServerRadiusSearch(lat, lng, radiusMiles) {
  var mySeq = ++pcRadiusSearchSeq;
  var catalog = document.querySelector('[data-category]');
  var category = (catalog && catalog.getAttribute('data-category')) || 'cats';
  var url = '/adoptions/pets/geo-filter.php?category=' + encodeURIComponent(category) +
    '&lat=' + encodeURIComponent(Number(lat).toFixed(2)) + '&lng=' + encodeURIComponent(Number(lng).toFixed(2)) +
    '&radius=' + encodeURIComponent(radiusMiles);
  return fetch(url)
    .then(function(r) { if (!r.ok) throw new Error('HTTP ' + r.status); return r.json(); })
    .then(function(data) {
      if (mySeq !== pcRadiusSearchSeq) return { ok: false, stale: true };
      if (!data || !Array.isArray(data.ids)) throw new Error('Unexpected response');
      pcRadiusMatchIds = data.ids;
      return { ok: true, pending: Number(data.pending) || 0 };
    })
    .catch(function() {
      if (mySeq === pcRadiusSearchSeq) pcRadiusMatchIds = null;
      return { ok: false, stale: mySeq !== pcRadiusSearchSeq };
    });
}

function pcParseLocation(loc) {
  if (!loc) return { city: '', state: '' };
  var text = String(loc).trim();
  var parts = text.split(',');
  var city = (parts[0] || '').trim();
  var state = (parts[1] || '').trim();
  // TASK-3d START — pet.location is free text typed by an admin. Tolerate a missing comma
  // ("Los Angeles CA") by peeling a trailing 2-letter code off as the state, and upper-case
  // short codes so "ca" / "Ca" / "CA" never become three separate State chips.
  if (!state) {
    var m = text.match(/^(.*\S)\s+([A-Za-z]{2})$/);
    if (m) { city = m[1].trim(); state = m[2]; }
  }
  if (state.length <= 3) state = state.toUpperCase();
  // TASK-3d END
  return { city: city, state: state };

function pcRunServerRadiusSearch(lat, lng, radiusMiles) {
  var mySeq = ++pcRadiusSearchSeq;

    '&radius=' + encodeURIComponent(radiusMiles);

// Free, no-API-key geocoding via OpenStreetMap Nominatim (usage policy caps
// requests at ~1/sec, so calls are chained through a single shared queue

function pcGeocodeLocation(query, countryCode) {
  var trimmed = (query || '').trim();
  if (!trimmed) return Promise.resolve(null);
  var key = trimmed.toLowerCase() + '|' + (countryCode || 'us'); // TASK-3b — cache key now includes country so the same text in two countries doesn't collide

function pcParseLocation(loc) {
  if (!loc) return { city: '', state: '' };
  var text = String(loc).trim();
  var parts = text.split(',');
  var city = (parts[0] || '').trim();
  var state = (parts[1] || '').trim();

  // ("Los Angeles CA") by peeling a trailing 2-letter code off as the state, and upper-case

  if (!state) {

    if (m) { city = m[1].trim(); state = m[2]; }
  }
  if (state.length <= 3) state = state.toUpperCase();

//    the smallest radius offered is 25 mi, so nothing is lost
var pcRadiusSearchSeq = 0;
function pcRunServerRadiusSearch(lat, lng, radiusMiles) {
  var mySeq = ++pcRadiusSearchSeq;
  var catalog = document.querySelector('[data-category]');
  var category = (catalog && catalog.getAttribute('data-category')) || 'cats';
  var url = '/adoptions/pets/geo-filter.php?category=' + encodeURIComponent(category) +
    '&lat=' + encodeURIComponent(Number(lat).toFixed(2)) + '&lng=' + encodeURIComponent(Number(lng).toFixed(2)) +
    '&radius=' + encodeURIComponent(radiusMiles);
  return fetch(url)
    .then(function(r) { if (!r.ok) throw new Error('HTTP ' + r.status); return r.json(); })
    .then(function(data) {
      if (mySeq !== pcRadiusSearchSeq) return { ok: false, stale: true };
      if (!data || !Array.isArray(data.ids)) throw new Error('Unexpected response');
      pcRadiusMatchIds = data.ids;
      return { ok: true, pending: Number(data.pending) || 0 };
    })
    .catch(function() {
      if (mySeq === pcRadiusSearchSeq) pcRadiusMatchIds = null;
      return { ok: false, stale: mySeq !== pcRadiusSearchSeq };
    });
}

function pcParseLocation(loc) {
  if (!loc) return { city: '', state: '' };
  var text = String(loc).trim();
  var parts = text.split(',');
  var city = (parts[0] || '').trim();
  var state = (parts[1] || '').trim();
  // TASK-3d START — pet.location is free text typed by an admin. Tolerate a missing comma
  // ("Los Angeles CA") by peeling a trailing 2-letter code off as the state, and upper-case
  // short codes so "ca" / "Ca" / "CA" never become three separate State chips.
  if (!state) {
    var m = text.match(/^(.*\S)\s+([A-Za-z]{2})$/);
    if (m) { city = m[1].trim(); state = m[2]; }
  }
  if (state.length <= 3) state = state.toUpperCase();
  // TASK-3d END
  return { city: city, state: state };

var pcRadiusSearchSeq = 0;
function pcRunServerRadiusSearch(lat, lng, radiusMiles) {
  var mySeq = ++pcRadiusSearchSeq;
  var catalog = document.querySelector('[data-category]');
  var category = (catalog && catalog.getAttribute('data-category')) || 'cats';
  var url = '/adoptions/pets/geo-filter.php?category=' + encodeURIComponent(category) +
    '&lat=' + encodeURIComponent(Number(lat).toFixed(2)) + '&lng=' + encodeURIComponent(Number(lng).toFixed(2)) +
    '&radius=' + encodeURIComponent(radiusMiles);
  return fetch(url)
    .then(function(r) { if (!r.ok) throw new Error('HTTP ' + r.status); return r.json(); })
    .then(function(data) {
      if (mySeq !== pcRadiusSearchSeq) return { ok: false, stale: true };

// Free, no-API-key geocoding via OpenStreetMap Nominatim (usage policy caps
// requests at ~1/sec, so calls are chained through a single shared queue

function pcGeocodeLocation(query, countryCode) {
  var trimmed = (query || '').trim();
  if (!trimmed) return Promise.resolve(null);
  var key = trimmed.toLowerCase() + '|' + (countryCode || 'us'); // TASK-3b — cache key now includes country so the same text in two countries doesn't collide

function pcParseLocation(loc) {
  if (!loc) return { city: '', state: '' };
  var text = String(loc).trim();
  var parts = text.split(',');
  var city = (parts[0] || '').trim();
  var state = (parts[1] || '').trim();

  // ("Los Angeles CA") by peeling a trailing 2-letter code off as the state, and upper-case

  if (!state) {

    if (m) { city = m[1].trim(); state = m[2]; }
  }
  if (state.length <= 3) state = state.toUpperCase();

// pet.json stores location as free text, e.g. "Los Angeles, CA"
function pcParseLocation(loc) {
  if (!loc) return { city: '', state: '' };
  var text = String(loc).trim();
  var parts = text.split(',');
  var city = (parts[0] || '').trim();
  var state = (parts[1] || '').trim();
  // TASK-3d START — pet.location is free text typed by an admin. Tolerate a missing comma
  // ("Los Angeles CA") by peeling a trailing 2-letter code off as the state, and upper-case
  // short codes so "ca" / "Ca" / "CA" never become three separate State chips.
  if (!state) {
    var m = text.match(/^(.*\S)\s+([A-Za-z]{2})$/);
    if (m) { city = m[1].trim(); state = m[2]; }
  }
  if (state.length <= 3) state = state.toUpperCase();
  // TASK-3d END
  return { city: city, state: state };
}

// TASK-3d START — second verification round hardening of the search helper:
//  - newest-search-wins token, so a slow earlier request can never overwrite a later one
//  - real failure handling: a missing/erroring endpoint no longer looks like "zero pets nearby"
//  - coordinates rounded to 2 decimals (~0.7 mi) before they go into the URL, because query

var pcRadiusSearchSeq = 0;
function pcRunServerRadiusSearch(lat, lng, radiusMiles) {
  var mySeq = ++pcRadiusSearchSeq;
  var catalog = document.querySelector('[data-category]');
  var category = (catalog && catalog.getAttribute('data-category')) || 'cats';
  var url = '/adoptions/pets/geo-filter.php?category=' + encodeURIComponent(category) +
    '&lat=' + encodeURIComponent(Number(lat).toFixed(2)) + '&lng=' + encodeURIComponent(Number(lng).toFixed(2)) +
    '&radius=' + encodeURIComponent(radiusMiles);
  return fetch(url)
    .then(function(r) { if (!r.ok) throw new Error('HTTP ' + r.status); return r.json(); })
    .then(function(data) {
      if (mySeq !== pcRadiusSearchSeq) return { ok: false, stale: true };
      if (!data || !Array.isArray(data.ids)) throw new Error('Unexpected response');
      pcRadiusMatchIds = data.ids;
      return { ok: true, pending: Number(data.pending) || 0 };
    })
    .catch(function() {

      return { ok: false, stale: mySeq !== pcRadiusSearchSeq };
    });
}

// Free, no-API-key geocoding via OpenStreetMap Nominatim (usage policy caps
// requests at ~1/sec, so calls are chained through a single shared queue
// rather than fired in parallel). Results are cached in memory for this page
// load only — a persistent shared cache is a Task 3b enhancement, not needed
// here since a single visitor rarely repeats the same search.
var pcGeocodeCache = {};
var pcGeocodeQueue = Promise.resolve();
var pcLastGeocodeAt = 0; // TASK-3d — timestamp of the last Nominatim call, so a lone search isn't delayed a full second for nothing

function pcGeocodeLocation(query, countryCode) {
  var trimmed = (query || '').trim();
  if (!trimmed) return Promise.resolve(null);
  var key = trimmed.toLowerCase() + '|' + (countryCode || 'us'); // TASK-3b — cache key now includes country so the same text in two countries doesn't collide
  if (Object.prototype.hasOwnProperty.call(pcGeocodeCache, key)) {
    return Promise.resolve(pcGeocodeCache[key]);
  }
  pcGeocodeQueue = pcGeocodeQueue.then(function() {
    return new Promise(function(resolve) {
      setTimeout(function() {
        pcLastGeocodeAt = Date.now(); // TASK-3d
        // TASK-3b — countrycodes now driven by the selected country (us/ca/gb/au)
        // instead of being hardcoded to "us", covering all 4 confirmed target
        // markets through this same single Nominatim integration.
        fetch('https://nominatim.openstreetmap.org/search?format=json&limit=1&countrycodes=' + (countryCode || 'us') + '&q=' + encodeURIComponent(query))

          .then(function(results) {

              ? { lat: parseFloat(results[0].lat), lng: parseFloat(results[0].lon) }

// pet.json stores location as free text, e.g. "Los Angeles, CA"
function pcParseLocation(loc) {
  if (!loc) return { city: '', state: '' };
  var text = String(loc).trim();
  var parts = text.split(',');
  var city = (parts[0] || '').trim();
  var state = (parts[1] || '').trim();
  // TASK-3d START — pet.location is free text typed by an admin. Tolerate a missing comma
  // ("Los Angeles CA") by peeling a trailing 2-letter code off as the state, and upper-case
  // short codes so "ca" / "Ca" / "CA" never become three separate State chips.
  if (!state) {
    var m = text.match(/^(.*\S)\s+([A-Za-z]{2})$/);
    if (m) { city = m[1].trim(); state = m[2]; }
  }
  if (state.length <= 3) state = state.toUpperCase();
  // TASK-3d END
  return { city: city, state: state };
}

// TASK-3c START — Calls geo-filter.php with the visitor's resolved
// coordinates + chosen radius; the server geocodes every pet in this
// category (using its own persisted cache) and returns just the IDs within
// range, nearest-first. Replaces Task 3b's pcGeocodeAllPetLocations(), which
// did this same work in the browser with a cache that reset every page load.
var pcRadiusMatchIds = null; // null = no server radius search active; array = active, filter to these IDs only
// TASK-3d START — second verification round hardening of the search helper:
//  - newest-search-wins token, so a slow earlier request can never overwrite a later one
//  - real failure handling: a missing/erroring endpoint no longer looks like "zero pets nearby"
//  - coordinates rounded to 2 decimals (~0.7 mi) before they go into the URL, because query
//    strings end up in server access logs and a visitor's exact position doesn't belong there;
//    the smallest radius offered is 25 mi, so nothing is lost
var pcRadiusSearchSeq = 0;
function pcRunServerRadiusSearch(lat, lng, radiusMiles) {
  var mySeq = ++pcRadiusSearchSeq;
  var catalog = document.querySelector('[data-category]');
  var category = (catalog && catalog.getAttribute('data-category')) || 'cats';
  var url = '/adoptions/pets/geo-filter.php?category=' + encodeURIComponent(category) +
    '&lat=' + encodeURIComponent(Number(lat).toFixed(2)) + '&lng=' + encodeURIComponent(Number(lng).toFixed(2)) +
    '&radius=' + encodeURIComponent(radiusMiles);
  return fetch(url)
    .then(function(r) { if (!r.ok) throw new Error('HTTP ' + r.status); return r.json(); })
    .then(function(data) {
      if (mySeq !== pcRadiusSearchSeq) return { ok: false, stale: true };
      if (!data || !Array.isArray(data.ids)) throw new Error('Unexpected response');
      pcRadiusMatchIds = data.ids;
      return { ok: true, pending: Number(data.pending) || 0 };
    })
    .catch(function() {
      if (mySeq === pcRadiusSearchSeq) pcRadiusMatchIds = null;
      return { ok: false, stale: mySeq !== pcRadiusSearchSeq };
    });
}
