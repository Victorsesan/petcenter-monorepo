## Task Number
TASK A2 — Listing schema expansion: relationships & flags

## What Was Required
Add two things to the pet-listing data model:

1. `grouped_with` — a bidirectional array field linking bonded-pair listings that must be adopted together, selectable from a dropdown (not free text) on the posting form, rendered on every browse page as one merged card with a "Bonded pair — must adopt together" label.
2. An automated **Needs Extra Love** flag — a scheduled daily check comparing each active listing's `last_inquiry_date` against its listed date; zero inquiries + over a threshold (21 days, adjustable constant) sets `needs_extra_love: true`, which pins the listing into the first page of default-sorted browse results and shows an SVG heart badge.

    function groupBondedPairs(list) {
      var byId = {};
      list.forEach(function(p) { byId[p.pet_id] = p; });
      var seen = {};
      var out = [];
      list.forEach(function(p) {
        if (seen[p.pet_id]) return;
        if (p.grouped_with && p.grouped_with.length) {
          var partner = byId[p.grouped_with[0]];
          if (partner && !seen[partner.pet_id]) {
            out.push({ bondedPair: [p, partner] });
            seen[p.pet_id] = true;
            seen[partner.pet_id] = true;
            return;
          }
        }
        out.push(p);
        seen[p.pet_id] = true;
      });
      return out;
    }

// TASK-A2 START — bonded-pair dropdown option sync + toggle handling.
// See the markup comment above (in addListingForm()'s template) for why

function refreshBondedPairOptions() {
  const keys = Object.keys(listingForms);
  keys.forEach(key => {
    const scroll = document.getElementById(`bonded_panel_scroll_${key}`);
    const hidden = document.querySelector(`input[data-a2-group-key="${key}"]`);
    const triggerLabel = document.getElementById(`bonded_wrap_${key}`)?.querySelector('.dropdown-trigger__label');
    if (!scroll) return; // card may have just been removed
    const others = keys.filter(k => k !== key);

    if (others.length === 0) {
      scroll.innerHTML = '<div style="padding:11px 16px;font-size:13px;color:var(--muted);">Add another pet above to pair with.</div>';
      if (hidden && hidden.value) { hidden.value = ''; if (triggerLabel) triggerLabel.textContent = 'Select a pet from this listing'; }
      return;
    }

    // If the previously-selected partner was removed, clear the stale selection.
    if (hidden && hidden.value && !others.includes(hidden.value)) {
      hidden.value = '';
      if (triggerLabel) triggerLabel.textContent = 'Select a pet from this listing';
    }

    scroll.innerHTML = others.map(k => {
      const otherNameField = listingForms[k].querySelector('[data-field="name"]');
      const label = (otherNameField && otherNameField.value.trim()) || `Pet ${k.split('_')[1]}`;
      const isActive = hidden && hidden.value === k ? ' is-active' : '';
      return `<div class="dropdown-panel__row${isActive}" data-value="${PC.esc(k)}">${PC.esc(label)}</div>`;
    }).join('');

    // Fresh nodes every refresh, so no wired-flag dedupe needed (unlike
    // initDropdowns(), whose rows are static and only wired once).
    scroll.querySelectorAll('.dropdown-panel__row').forEach(row => {
      row.addEventListener('click', () => {
        scroll.querySelectorAll('.dropdown-panel__row').forEach(r => r.classList.remove('is-active'));
        row.classList.add('is-active');
        if (hidden) hidden.value = row.getAttribute('data-value');
        if (triggerLabel) triggerLabel.textContent = row.textContent;
        closeAllDropdowns();
      });
    });
  });
}

function toggleBondedPairField(key) {
  const wrap = document.getElementById(`bonded_wrap_${key}`);
  const isChecked = document.getElementById(`bonded_toggle_${key}`).checked;
  wrap.style.display = isChecked ? 'block' : 'none';
  const hidden = document.querySelector(`input[data-a2-group-key="${key}"]`);
  if (!isChecked) {
    // Unchecking clears the selection rather than silently keeping a
    // pairing the merchant no longer wants applied on submit.
    if (hidden) hidden.value = '';
    const triggerLabel = wrap.querySelector('.dropdown-trigger__label');
    if (triggerLabel) triggerLabel.textContent = 'Select a pet from this listing';
  } else {
    refreshBondedPairOptions();
  }
}
